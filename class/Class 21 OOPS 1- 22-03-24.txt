class student{
    constructor(name,age,grade)
    {
        this.name=name
        this.age=age
        this.grade=grade
    }

    displaydetails()
    {
        console.log("name",this.name,"age",this.age,"& grade",this.grade)
    }

    promote()
    {
        this.grade++
        console.log(this.grade)
    }
}

let student2= new student("kiran",22,6);
